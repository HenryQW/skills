# Examples

  # Download and run the Copilot CLI
  $ gh copilot

  # Run the Copilot CLI
  $ gh copilot -p "Summarize this week's commits" --allow-tool 'shell(git)'

  # Remove the Copilot CLI (if installed through gh)
  $ gh copilot --remove

  # Run the Copilot CLI help command
  $ gh copilot -- --help
